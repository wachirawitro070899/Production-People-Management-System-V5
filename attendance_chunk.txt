function attendancePie(title,subtitle,stats){const vals=[stats.onTime||0,stats.late||0,stats.personal||0,stats.sick||0,stats.absent||0],labels=['ตรงเวลา','มาสาย','ลากิจ','ลาป่วย','ขาดงาน'],colors=['#22c55e','#f59e0b','#8b5cf6','#38bdf8','#ef4444'],total=vals.reduce((a,b)=>a+b,0);let acc=0,parts=[];for(let i=0;i<vals.length;i++){const p=total?vals[i]/total*100:0,from=acc,to=acc+p;parts.push(`${colors[i]} ${from.toFixed(2)}% ${to.toFixed(2)}%`);acc=to}const style=total?`background:conic-gradient(${parts.join(',')})`:'background:#e5e7eb';return `<div class="panel attendance-pie-card"><h3>${esc(title)}</h3><small>${esc(subtitle||'')}</small><div class="attendance-pie-wrap"><div class="attendance-pie" style="${style}"><div><b>${total}</b><span>รายการ</span></div></div><div class="attendance-pie-legend">${labels.map((l,i)=>`<div><i style="background:${colors[i]}"></i><span>${l}</span><b>${vals[i]}</b></div>`).join('')}</div></div></div>`}
function attendanceMonthlyTrend(rows){const months=Array.from({length:12},(_,i)=>({m:i+1,late:0,leave:0,absent:0,deduct:0}));for(const x of rows)for(const r of x.rows){const m=Number(String(r.date||'').slice(5,7));if(!m||!months[m-1])continue;months[m-1].late+=attendanceStatus(r)==='Late'||r.exception?.type==='late'?1:0;months[m-1].leave+=r.exception?.type==='leave'?1:0;months[m-1].absent+=!r.checkIn&&r.exception?.type==='absent'?1:0;months[m-1].deduct+=attendanceDeduction(r)}const max=Math.max(1,...months.map(x=>x.late+x.leave+x.absent));return `<div class="panel attendance-monthly"><h3>แนวโน้มรายเดือน</h3><div class="attendance-month-grid">${months.map(x=>`<div class="attendance-month"><b>${String(x.m).padStart(2,'0')}</b><div class="attendance-month-bar"><i style="height:${Math.round((x.late+x.leave+x.absent)/max*100)}%"></i></div><small>สาย ${x.late} • ลา ${x.leave} • ขาด ${x.absent}<br>หัก ${x.deduct}</small></div>`).join('')}</div></div>`}
function attendanceMonthlyChecksheet(emp,year,month){
 if(!emp)return `<div class="panel attendance-checksheet"><h3>เช็คชีท ขาด / ลา / มาสาย รายเดือน</h3><p class="modal-note">เลือกพนักงานเพื่อดูเช็คชีทรายเดือน</p></div>`;
 const m=Math.max(1,Math.min(12,Number(month)||new Date().getMonth()+1)),prefix=`${year}-${String(m).padStart(2,'0')}-`,days=new Date(Number(year),m,0).getDate(),today=thaiDateKey(),monthNames=['มกราคม','กุมภาพันธ์','มีนาคม','เมษายน','พฤษภาคม','มิถุนายน','กรกฎาคม','สิงหาคม','กันยายน','ตุลาคม','พฤศจิกายน','ธันวาคม'];
 let late=0,personal=0,sick=0,absent=0,onTime=0,present=0,deduct=0;
 const rows=Array.from({length:days},(_,i)=>{const day=i+1,date=prefix+String(day).padStart(2,'0'),rec=attendanceFor(emp.id,date),holiday=isHoliday(date),future=date>today,weekday=new Date(Number(year),m-1,day).toLocaleDateString('th-TH',{weekday:'short'});let kind='-',reason='-',checkIn='-',marks={ontime:'',late:'',personal:'',sick:'',absent:''},points=0;
  if(holiday){kind='วันหยุด';reason=typeof holidays[date]==='object'?(holidays[date]?.name||'วันหยุด'):(holidays[date]||'วันหยุด');}
  else if(attendanceIsFullCleanupDate(date,emp.section)){kind='ช่วงทดลองระบบ';reason='ล้างข้อมูลแล้ว / ไม่นับคะแนน';}
  else if(rec?.exception?.type==='leave'){if(rec.exception.leaveType==='sick'){kind='ลาป่วย';marks.sick='✓';sick++}else{kind='ลากิจ';marks.personal='✓';personal++}reason=rec.exception?.reason||'-';points=attendanceDeduction(rec);}
  else if(!rec?.checkIn&&rec?.exception?.type==='absent'){kind='ขาดงาน';marks.absent='✓';absent++;reason=rec.exception?.reason||'ไม่มีเช็คชื่อ';points=attendanceDeduction(rec);}
  else if(rec?.checkIn){present++;checkIn=thaiTime(new Date(rec.checkIn));if(attendanceStatus(rec)==='Late'||rec.exception?.type==='late'){kind='มาสาย';marks.late='✓';late++;reason=rec.lateReason?.reason||rec.lateNotice?.reason||'-'}else{kind='ตรงเวลา';marks.ontime='✓';onTime++}points=attendanceDeduction(rec);}
  else if(attendanceIsTrialDate(date,emp.section)){kind='ช่วงทดลองระบบ';reason='ไม่นับคะแนน';}
  else if(future){kind='รอวันทำงาน';}
  else if(date===today){kind='วันนี้ / รอข้อมูล';}
  deduct+=points;
  return `<tr class="${holiday?'checksheet-holiday':''}"><td>${day}</td><td>${esc(weekday)}</td><td>${checkIn}</td><td class="check-cell ok">${marks.ontime}</td><td class="check-cell warn">${marks.late}</td><td class="check-cell leave">${marks.personal}</td><td class="check-cell sick">${marks.sick}</td><td class="check-cell absent">${marks.absent}</td><td>${esc(kind)}</td><td>${esc(reason)}</td><td>${points?'-'+points:'0'}</td></tr>`;
 }).join('');
 return `<div class="panel attendance-checksheet"><div class="checksheet-head"><div><h3>เช็คชีท ขาด / ลา / มาสาย รายเดือน</h3><p><b>${esc(emp.id)} · ${esc(emp.name)}</b> • ${esc(emp.section||'')} • ${monthNames[m-1]} ${year}</p></div><div class="checksheet-summary"><span>ตรงเวลา <b>${onTime}</b></span><span>สาย <b>${late}</b></span><span>ลากิจ <b>${personal}</b></span><span>ลาป่วย <b>${sick}</b></span><span>ขาด <b>${absent}</b></span><span>หัก <b>-${deduct}</b></span></div></div><div class="table-wrap"><table class="monthly-checksheet-table"><thead><tr><th>วันที่</th><th>วัน</th><th>เข้า</th><th>ตรงเวลา</th><th>สาย</th><th>ลากิจ</th><th>ลาป่วย</th><th>ขาด</th><th>สถานะ</th><th>เหตุผล/หมายเหตุ</th><th>หัก</th></tr></thead><tbody>${rows}</tbody></table></div><p class="modal-note">เครื่องหมาย ✓ แสดงสถานะของพนักงานในแต่ละวัน ข้อมูลใช้ชุดเดียวกับ Attendance บน Firebase จึงเห็นตรงกันทุกเครื่องเมื่อซิงก์สำเร็จ</p></div>`
}

function attendanceMonthlyEmployeeSummary(emp,year,month){
 const m=Math.max(1,Math.min(12,Number(month)||new Date().getMonth()+1)),prefix=`${year}-${String(m).padStart(2,'0')}-`,rows=mergeAttendanceRecords([],attendance.filter(r=>sameAttendanceEmployeeId(r.employeeId,emp.id)&&String(r.date||'').startsWith(prefix)&&!isHoliday(r.date)));
 let onTime=0,late=0,personal=0,sick=0,absent=0,present=0,deduct=0;
 for(const r of rows){
  if(r.checkIn){present++;if(attendanceStatus(r)==='Late'||r.exception?.type==='late')late++;else onTime++;}
  else if(r.exception?.type==='leave'){if(r.exception?.leaveType==='sick')sick++;else personal++;}
  else if(r.exception?.type==='absent')absent++;
  deduct+=attendanceDeduction(r);
 }
 return{emp,present,onTime,late,personal,sick,absent,deduct,events:onTime+late+personal+sick+absent};
}

function attendanceDateRangeKeys(start,end){
 const out=[],a=new Date(String(start)+'T00:00:00+07:00'),b=new Date(String(end)+'T00:00:00+07:00');
 if(Number.isNaN(a.getTime())||Number.isNaN(b.getTime())||a>b)return out;
 for(let d=new Date(a);d<=b;d.setDate(d.getDate()+1))out.push(`${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`);
 return out;
}
